```js
// Section-scoped plain `Label: value` replace inside `## Session Continuity`.
function sessionReplacePlainField(content, label, newValue) {
  const m = content.match(/^## Session Continuity[ \t]*$/m); if (!m) return null;
  const start = m.index + m[0].length;
  const rest = content.slice(start); const nextH = rest.search(/^## /m);
  const end = nextH === -1 ? content.length : start + nextH;
  const section = content.slice(start, end);
  const esc = label.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const re = new RegExp(`^(${esc}:[ \\t]*)(.*)$`, 'im');
  const hit = section.match(re); if (!hit) return null;
  const wrapped = /^`[^`]*`$/.test(hit[2].trim());
  const val = wrapped ? '`' + newValue + '`' : newValue;
  return content.slice(0, start) + section.replace(re, `$1${val}`) + content.slice(end);
}
```

<gotchas>
- The section heading may be followed by a blank line. The regex is anchored per-line with `m`.
- `Last session:` values in the real file contain em dashes and backticks. The replace swaps the entire value, so do not try to preserve partial content.
- The `$` in a replacement string is special: newValue could contain `$&`. Use a replacer function (`section.replace(re, (_, p1) => p1 + val)`) to avoid `$` interpolation.
</gotchas>
