Detector (define in the test file; there is no new lib module):
```js
function findRgFlagMisuse(text) {
  const out = [];
  text.split('\n').forEach((line, i) => {
    for (const m of line.matchAll(/(?:^|[\s`|;&(])rg\s+([^|;&`\n]*)/g)) {
      const bad = m[1].split(/\s+/).find(t => /^-[A-Za-z]*E[A-Za-z]*$/.test(t));   // single dash, not --long
      if (bad) { out.push({ line: i + 1, text: line.trim(), token: bad }); break; }
    }
  });
  return out;
}
```
