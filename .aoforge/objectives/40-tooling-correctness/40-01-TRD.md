Current getMilestoneInfo (roadmap.cjs:12-24):
```js
function getMilestoneInfo(cwd) {
  try {
    const roadmap = fs.readFileSync(path.join(cwd, '.planning', 'ROADMAP.md'), 'utf-8');
    const versionMatch = roadmap.match(/v(\d+\.\d+)/);
    const nameMatch = roadmap.match(/## .*v\d+\.\d+[:\s]+([^\n(]+)/);
    return { version: versionMatch ? versionMatch[0] : 'v1.0', name: nameMatch ? nameMatch[1].trim() : 'milestone' };
  } catch { return { version: 'v1.0', name: 'milestone' }; }
}
```
Keep the two legacy regexes verbatim as the no-`## Milestones` fallback.
